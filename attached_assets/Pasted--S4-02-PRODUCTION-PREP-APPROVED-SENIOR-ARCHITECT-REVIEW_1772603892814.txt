def get_tickets():
    tenant_id = request.args.get('tenant_id', 'default')
    conn = sqlite3.connect('data/ndr.db')
    if tenant_id == 'all':
        rows = conn.execute("SELECT * FROM 'ndr-tickets' ORDER BY timestamp DESC LIMIT 100").fetchall()
    else:
        rows = conn.execute(
            "SELECT * FROM 'ndr-tickets' WHERE tenant_id=? ORDER BY timestamp DESC LIMIT 100",
            (tenant_id,)
        ).fetchall()
    conn.close()
    return jsonify({"tickets": [dict(r) for r in rows], "tenant_id": tenant_id, "count": len(rows)})
